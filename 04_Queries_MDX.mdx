-- ---------------------------------------------------------------------
-- MDX 1: Phân tích Doanh thu thuần, Biên đóng góp và Tỷ lệ biên lợi nhuận
-- theo Ngành hàng (Category) và Thương hiệu (Brand)
-- Kỹ thuật: Calculated Member, NonEmpty, CrossJoin
-- ---------------------------------------------------------------------
WITH 
MEMBER [Measures].[Contribution Margin Rate] AS
    IIF(
        [Measures].[Net Revenue] = 0 OR ISEMPTY([Measures].[Net Revenue]),
        NULL,
        [Measures].[Contribution Margin] / [Measures].[Net Revenue]
    ), FORMAT_STRING = "Percent"

SELECT 
    {
        [Measures].[Net Revenue],
        [Measures].[Contribution Margin],
        [Measures].[Contribution Margin Rate]
    } ON COLUMNS,
    NON EMPTY 
    ORDER(
        [Dim Product].[Category].[Category].Members * 
        [Dim Product].[Brand].[Brand].Members,
        [Measures].[Contribution Margin],
        BDESC
    ) ON ROWS
FROM [Ecommerce DW];


-- ---------------------------------------------------------------------
-- MDX 2: Phân tích xu hướng thời gian & Tăng trưởng doanh thu theo Quý (QoQ Growth)
-- Kỹ thuật: ParallelPeriod, Time Intelligence, IIF kiểm tra chia cho 0
-- ---------------------------------------------------------------------
WITH 
MEMBER [Measures].[Prev Quarter Revenue] AS
    (
        ParallelPeriod(
            [Dim Date].[Calendar].[Quarter Name],
            1,
            [Dim Date].[Calendar].CurrentMember
        ),
        [Measures].[Net Revenue]
    ), FORMAT_STRING = "Currency"

MEMBER [Measures].[QoQ Growth Rate] AS
    IIF(
        IsEmpty([Measures].[Prev Quarter Revenue]) OR [Measures].[Prev Quarter Revenue] = 0,
        NULL,
        ([Measures].[Net Revenue] - [Measures].[Prev Quarter Revenue]) / [Measures].[Prev Quarter Revenue]
    ), FORMAT_STRING = "Percent"

SELECT 
    {
        [Measures].[Net Revenue],
        [Measures].[Prev Quarter Revenue],
        [Measures].[QoQ Growth Rate]
    } ON COLUMNS,
    NON EMPTY 
    [Dim Date].[Calendar].[Quarter Name].Members ON ROWS
FROM [Ecommerce DW];


-- ---------------------------------------------------------------------
-- MDX 3: Tỷ lệ giao hàng đúng hạn (On-Time Delivery Rate) theo Kho & Phương thức vận chuyển
-- Kỹ thuật: Measure Group Fact Order Fulfillment, Phép tính tỷ lệ
-- ---------------------------------------------------------------------
WITH 
MEMBER [Measures].[On-Time Rate] AS
    IIF(
        [Measures].[Fact Order Fulfillment Count] = 0,
        NULL,
        1.0 - ([Measures].[Late Flag] / [Measures].[Fact Order Fulfillment Count])
    ), FORMAT_STRING = "Percent"

MEMBER [Measures].[Avg Delay Days] AS
    IIF(
        [Measures].[Fact Order Fulfillment Count] = 0,
        NULL,
        [Measures].[Delivery Variance Days] / [Measures].[Fact Order Fulfillment Count]
    ), FORMAT_STRING = "#,##0.0"

SELECT 
    {
        [Measures].[Fact Order Fulfillment Count],
        [Measures].[Late Flag],
        [Measures].[On-Time Rate],
        [Measures].[Avg Delay Days]
    } ON COLUMNS,
    NON EMPTY 
    [Dim Shipping].[Shipping Method].[Shipping Method].Members *
    [Dim Shipping].[Warehouse].[Warehouse].Members ON ROWS
FROM [Ecommerce DW];


-- ---------------------------------------------------------------------
-- MDX 4: Top 5 Kênh Marketing mang lại doanh thu đơn hàng đầu tiên cao nhất
-- Kỹ thuật: TopCount, Measure Group Fact Customer Acquisition
-- ---------------------------------------------------------------------
WITH 
MEMBER [Measures].[Avg First Order Value] AS
    IIF(
        [Measures].[New Customer Count] = 0,
        NULL,
        [Measures].[First Order Net Revenue] / [Measures].[New Customer Count]
    ), FORMAT_STRING = "Currency"

SELECT 
    {
        [Measures].[New Customer Count],
        [Measures].[First Order Net Revenue],
        [Measures].[Customer Acquisition Cost],
        [Measures].[Avg First Order Value]
    } ON COLUMNS,
    TOPCOUNT(
        [Dim Marketing Campaign].[Marketing Channel].[Marketing Channel].Members,
        5,
        [Measures].[First Order Net Revenue]
    ) ON ROWS
FROM [Ecommerce DW];


-- ---------------------------------------------------------------------
-- MDX 5: Phân tích tỷ lệ khách hàng rủi ro rời bỏ (Churn Risk) theo Hạng thành viên
-- Kỹ thuật: Fact Customer Snapshot, Lọc phần tử slice theo ngày Snapshot cuối cùng
-- ---------------------------------------------------------------------
WITH 
MEMBER [Measures].[Active Rate] AS
    IIF(
        [Measures].[Customer Count] = 0,
        NULL,
        [Measures].[Active 90d Flag] / [Measures].[Customer Count]
    ), FORMAT_STRING = "Percent"

MEMBER [Measures].[Churn Risk Rate] AS
    IIF(
        [Measures].[Customer Count] = 0,
        NULL,
        1.0 - ([Measures].[Active 90d Flag] / [Measures].[Customer Count])
    ), FORMAT_STRING = "Percent"

SELECT 
    {
        [Measures].[Customer Count],
        [Measures].[Active 90d Flag],
        [Measures].[Active Rate],
        [Measures].[Churn Risk Rate],
        [Measures].[Cumulative Net Revenue]
    } ON COLUMNS,
    NON EMPTY 
    [Dim Customer].[Customer Segment].[Customer Segment].Members ON ROWS
FROM [Ecommerce DW]
WHERE ([Dim Date].[Calendar].[Year].[2025].[Q4].[December]);


-- ---------------------------------------------------------------------
-- MDX 6: Khoan sâu địa lý (Drill-Down Geography) từ Quốc gia xuống Vùng
-- Kỹ thuật: Phân cấp địa lý Dim_Region (Country -> Region -> State)
-- ---------------------------------------------------------------------
SELECT 
    {
        [Measures].[Net Revenue],
        [Measures].[Gross Sales],
        [Measures].[Discount Amount]
    } ON COLUMNS,
    NON EMPTY 
    DRILLDOWNLEVEL(
        [Dim Region].[Administrative].[Country].Members
    ) ON ROWS
FROM [Ecommerce DW];
